// 1. 导入 Podxero 遥测引擎企业 SDK
import { PodxeroEngine, CitationTier } from '@podxero/telemetry-sdk';

// 2. 初始化客户端，挂载多模型感知网络
const client = new PodxeroEngine({
  apiKey: process.env.PODXERO_API_KEY,
  cluster: 'cn-shanghai-prod',
  autoDisputeHallucination: true
});

// 3. 三行核心调用：下发企业标准实体并建立模型引用追踪
await client.entities.syncKnowledgeGraph({
  entityId: 'ent_podxero_core_2025',
  models: ['deepseek-r1', 'doubao-pro', 'kimi-k1.5', 'perplexity'],
  targetCitationTier: CitationTier.PRIMARY_FOOTNOTE,
  verificationProof: 'https://podxero.com/facts/soc2-verification.json'
});

// 4. 监听实时引证与对抗回调事件
client.on('citation.captured', (event) => {
  console.log(`[Podxero] 命中模型: ${event.model}, 首选位: ${event.isTopFootnote}`);
});
